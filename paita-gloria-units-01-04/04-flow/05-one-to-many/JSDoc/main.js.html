/**
 * @function oneToMany
 * @description This function takes a number and a singular noun as arguments. If the number is 0 or greater than 1, it returns the plural form of the noun; otherwise, it returns the singular form.
 * 
 * The function includes predefined pluralization rules for common irregular nouns such as:
 * - "cat" → "cats"
 * - "sheep" → "sheep" (unchanged)
 * - "goose" → "geese"
 * 
 * If the noun is not in the predefined list, the function defaults to adding "s" to the singular form.
 * 
 * @param {number} num - The quantity associated with the noun (must be a positive number).
 * @param {string} noun - A name in singular form to be turned into its plural form if necessary.
 * @returns {string} A string containing the number and the correctly pluralized form of the noun (e.g., "5 cats", "1 dog").
 * 
 * @example
 * console.log(oneToMany(1, "cat")); 
 * // Logs: "1 cat"
 * 
 * @example
 * console.log(oneToMany(3, "dog")); 
 * // Logs: "3 dogs"
 * 
 * @example
 * console.log(oneToMany(4, "goose")); 
 * // Logs: "4 geese"
 * 
 * @example
 * console.log(oneToMany(0, "bird")); 
 * // Logs: "0 birds"
 */

function oneToMany(num, noun) {

    //Checks if the number is equal to 0 or greater than 1
    if (num == 0 || num > 1) {

        let plural;
        let pluralResult;

        //Assigns the corresponding plural form the the variable
        switch (noun) {
            case "cat":
                plural = "cats";
                break;
            case "dog":
                plural = "dogs";
                break;
            case "sheep":
                plural = "sheep";
                break;
            case "goose":
                plural = "geese";
                break;
            default:
                plural = `${noun}s`;
        }

        pluralResult = `${num} ${plural}`;
        return pluralResult;

    //If the number is 1
    } else {
        return `${num} ${noun}`;
    }

}

// Test Cases
console.log(oneToMany(1, "cat"));     // "1 cat"
console.log(oneToMany(1, "dog"));     // "1 dog"
console.log(oneToMany(1, "goose"));   // "1 goose"
console.log(oneToMany(1, "sheep"));   // "1 sheep"

console.log(oneToMany(2, "cat"));     // "2 cats"
console.log(oneToMany(3, "dog"));     // "3 dogs"
console.log(oneToMany(4, "goose"));   // "4 geese"
console.log(oneToMany(5, "sheep"));   // "5 sheep"
console.log(oneToMany(0, "bird"));    // "0 birds"
console.log(oneToMany(0, "slug"));    // "0 slugs"

// Edge Cases
console.log(oneToMany(5, ""));    // "5 s"
console.log(oneToMany(1, ""));    // "1 "
console.log(oneToMany(5, "Cat"));    // "5 Cats"
console.log(oneToMany(5, "Goose"));    // "5 Gooses"
