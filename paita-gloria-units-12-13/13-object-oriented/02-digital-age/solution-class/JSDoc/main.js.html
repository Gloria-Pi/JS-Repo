/**
 * Creates a Video object with a title and duration in seconds.
 *
 * @class
 * @param {Object} [config={}] - The configuration object.
 * @param {string} [config.title] - The title of the video. Defaults to "Title" if missing or invalid.
 * @param {number} [config.seconds] - The duration of the video in seconds. Must be non-negative. Defaults to 0.
 */
class Video {
    constructor(config = {}) {
        // Title must be a string and non-empty if provided
        if (typeof config.title !== "undefined" && typeof config.title !== "string") {
            console.log("Invalid title.");
            this.title = "Title";
        } else {
            this.title = config.title || "Title";
        }

        // Seconds must be a non-negative number
        this.seconds = (
            typeof config.seconds === "number" && config.seconds >= 0) ? config.seconds : 0;
    }
}

/**
 * Logs how many seconds of the video were watched.
 *
 * @function
 * @name Video#watch
 * @param {number} [nSeconds] - The number of seconds watched. If omitted, assumes the full video duration.
 *
 * @example
 * const vid = new Video({ title: "Sample", seconds: 100 });
 * vid.watch(30); // You watched 30 seconds of "Sample"
 * vid.watch();   // You watched all 100 seconds of "Sample"
 */
Video.prototype.watch = function (nSeconds) {
    if (nSeconds !== undefined) {
        if (typeof nSeconds !== "number" || nSeconds < 0) {
            console.log("The number you've inputted is invalid.");
        } else {
            console.log(`You watched ${nSeconds} seconds of "${this.title}"`);
        }
    } else {
        console.log(`You watched all ${this.seconds} seconds of "${this.title}"`);
    }
};

/**
 * Creates a MusicVideo object that extends Video, adding an artist property.
 *
 * @class
 * @extends Video
 * @param {Object} [config={}] - The configuration object.
 * @param {string} [config.title] - The title of the music video. Defaults to "Title" if missing or invalid.
 * @param {number} [config.seconds] - The duration in seconds. Must be non-negative. Defaults to 0.
 * @param {string} [config.artist] - The artist's name. Defaults to "Artist" if missing or invalid.
 */
class MusicVideo extends Video {
    constructor(config = {}) {
        super(config);

        // Artist must be a string if provided
        if (typeof config.artist !== "undefined" && typeof config.artist !== "string") {
            console.log("Invalid artist name.");
            this.artist = "Artist";
        } else {
            this.artist = config.artist || "Artist";
        }
    }
}

/**
 * Logs a message indicating the music video has been played.
 *
 * @function
 * @name MusicVideo#play
 *
 * @example
 * const mv = new MusicVideo({ title: "My Song", artist: "The Artist" });
 * mv.play(); // You played "My Song" by "The Artist"
 */
MusicVideo.prototype.play = function () {
    console.log(`You played "${this.title}" by "${this.artist}"`);
};

    }
